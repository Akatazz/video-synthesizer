cmake_minimum_required(VERSION 3.20)
project(VideoSynthesizer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4>=4.10)
find_package(nlohmann_json 3.10 REQUIRED)
pkg_check_modules(GSTREAMER REQUIRED IMPORTED_TARGET gstreamer-app-1.0 gstreamer-fft-1.0)
add_library(synth-core INTERFACE)
target_include_directories(synth-core INTERFACE src)
target_link_libraries(synth-core INTERFACE nlohmann_json::nlohmann_json PkgConfig::GSTREAMER ${CMAKE_DL_LIBS})
add_executable(video-synthesizer src/main.cpp)
target_link_libraries(video-synthesizer PRIVATE synth-core PkgConfig::GTK4)
target_compile_options(video-synthesizer PRIVATE -Wall -Wextra -Werror)
include(CTest)
add_executable(synth-checks tests/checks.cpp)
target_link_libraries(synth-checks PRIVATE synth-core)
target_compile_options(synth-checks PRIVATE -Wall -Wextra -Werror)
add_test(NAME native-contract-and-renderer COMMAND synth-checks "${CMAKE_CURRENT_SOURCE_DIR}/../../contracts/presets/classic-feedback.json")
install(TARGETS video-synthesizer RUNTIME DESTINATION bin)
